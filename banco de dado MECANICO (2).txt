CREATE TABLE mecanico 
( 
 idmecanico INT PRIMARY KEY AUTO_INCREMENT,
 cpf VARCHAR(20) UNIQUE NOT NULL,
 nome VARCHAR(100) NOT NULL,
 telefone VARCHAR(20) NOT NULL,
 data_admissao DATE NOT NULL
); 

CREATE TABLE cliente 
( 
 idcliente INT PRIMARY KEY AUTO_INCREMENT,
 cpf VARCHAR(20) UNIQUE NOT NULL,
 nome VARCHAR(100) NOT NULL,
 telefone VARCHAR(20),
 email VARCHAR(100)
); 

CREATE TABLE veiculo 
( 
 placa VARCHAR(10) PRIMARY KEY,
 idcliente INT NOT NULL,
 marca VARCHAR(50),
 modelo VARCHAR(50),
 ano INT,
 cor VARCHAR(30),
 FOREIGN KEY (idcliente) REFERENCES cliente(idcliente)
); 

CREATE TABLE servicos 
( 
 codservico INT PRIMARY KEY AUTO_INCREMENT,
 nome VARCHAR(100) NOT NULL,
 descricao VARCHAR(255),
 tempo_medio INT
); 

CREATE TABLE ordem_de_servico 
( 
 idordem INT PRIMARY KEY AUTO_INCREMENT,
 idveiculo VARCHAR(10) NOT NULL,
 valor DECIMAL(10,2),
 status VARCHAR(50),
 problemarelatado VARCHAR(255),
 FOREIGN KEY (idveiculo) REFERENCES veiculo(placa)
); 

CREATE TABLE servicos_mecanicos 
( 
 codserv_mec INT PRIMARY KEY AUTO_INCREMENT,
 data_inicio DATE NOT NULL,
 data_saida DATE,
 valorunitario DECIMAL(10,2) NOT NULL,
 status VARCHAR(50) NOT NULL,
 idmecanico INT,
 codservico INT,
 idordem INT,
 FOREIGN KEY (idmecanico) REFERENCES mecanico(idmecanico),
 FOREIGN KEY (codservico) REFERENCES servicos(codservico),
 FOREIGN KEY (idordem) REFERENCES ordem_de_servico(idordem)
);
